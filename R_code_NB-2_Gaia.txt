## 4b) Profile likelihood CI for global mu
nb2_loglik_int <- function(mu, theta, y) {
  if (mu <= 0 || theta <= 0) return(-Inf)
  term1 <- lgamma(y + theta) - lgamma(theta) - lgamma(y + 1)
  term2 <- theta * log(theta / (theta + mu))
  term3 <- y * log(mu / (theta + mu))
  sum(term1 + term2 + term3)
}

profile_theta_for_mu <- function(mu0, y, theta_start) {
  fn_neg <- function(log_theta) {
    theta <- exp(log_theta)
    -nb2_loglik_int(mu0, theta, y)
  }

  opt <- optim(
    par    = log(theta_start),
    fn     = fn_neg,
    method = "Brent",
    lower  = log(1e-6),
    upper  = log(1e6)
  )

  list(theta_hat_mu = exp(opt$par),
       loglik_mu    = -opt$value,
       convergence  = opt$convergence)
}

dev_mu_int <- function(mu0, y, theta_start, LL_max) {
  if (mu0 <= 0) return(Inf)
  prof <- profile_theta_for_mu(mu0, y, theta_start = theta_start)
  2 * (LL_max - prof$loglik_mu)
}

cutoff <- qchisq(0.95, df = 1)

lower_fun_int <- function(mu0) {
  dev_mu_int(mu0, y, theta_hat_int, LL_max_int) - cutoff
}
upper_fun_int <- function(mu0) {
  dev_mu_int(mu0, y, theta_hat_int, LL_max_int) - cutoff
}

radec_to_gal <- function(ra_deg, dec_deg) {
  d2r <- pi / 180
  ra  <- ra_deg  * d2r
  dec <- dec_deg * d2r

  x_eq <- cos(dec) * cos(ra)
  y_eq <- cos(dec) * sin(ra)
  z_eq <- sin(dec)

  ## IAU J2000 rotation matrix: equatorial -> Galactic
  R <- matrix(c(
    -0.0548755604162154, -0.8734370902348850, -0.4838350155487132,
     0.4941094278755837, -0.4448296299600112,  0.7469822444972189,
    -0.8676661490190047, -0.1980763734312015,  0.4559837761750669
  ), nrow = 3, byrow = TRUE)

  x_gal <- R[1,1] * x_eq + R[1,2] * y_eq + R[1,3] * z_eq
  y_gal <- R[2,1] * x_eq + R[2,2] * y_eq + R[2,3] * z_eq
  z_gal <- R[3,1] * x_eq + R[3,2] * y_eq + R[3,3] * z_eq

  b_rad <- asin(z_gal)
  l_rad <- atan2(y_gal, x_gal)
  l_rad <- ifelse(l_rad < 0, l_rad + 2*pi, l_rad)

  data.frame(
    l = l_rad / d2r,
    b = b_rad / d2r
  )
}

nb2_loglik_reg_full <- function(beta0, beta1, theta, y, abs_b) {
  if (theta <= 0) return(-Inf)

  eta <- beta0 + beta1 * abs_b
  mu  <- exp(eta)

  if (any(!is.finite(mu))) return(-Inf)

  term1 <- lgamma(y + theta) - lgamma(theta) - lgamma(y + 1)
  term2 <- theta * log(theta / (theta + mu))
  term3 <- y * log(mu / (theta + mu))
  sum(term1 + term2 + term3)
}

profile_loglik_mu_reg <- function(mu0, b0, y, abs_b,
                                  beta1_start, theta_start) {
  if (mu0 <= 0) return(list(loglik_mu = -Inf, convergence = NA_integer_))

  fn_neg <- function(par) {
    beta1 <- par[1]
    theta <- exp(par[2])
    beta0 <- log(mu0) - beta1 * abs(b0)
    -nb2_loglik_reg_full(beta0, beta1, theta, y, abs_b)
  }

  opt <- optim(
    par    = c(beta1_start, log(theta_start)),
    fn     = fn_neg,
    method = "BFGS",
    control = list(maxit = 1000)
  )

  list(
    beta1_hat_mu = opt$par[1],
    theta_hat_mu = exp(opt$par[2]),
    loglik_mu = -opt$value,
    convergence = opt$convergence
  )
}

dev_mu_reg <- function(mu0, b0) {
  prof <- profile_loglik_mu_reg(
    mu0 = mu0,
    b0 = b0,
    y = y,
    abs_b = gaia_counts$abs_b,
    beta1_start = beta_hat_reg[2],
    theta_start = theta_hat_reg
  )

  if (!is.finite(prof$loglik_mu)) return(Inf)
  2 * (LL_max_reg - prof$loglik_mu)
}

mollweide <- function(lon_deg, lat_deg) {
  lon <- lon_deg * pi / 180
  lat <- lat_deg * pi / 180

  lon <- (lon + pi) %% (2 * pi) - pi

  theta <- lat
  for (kk in 1:20) {
    step <- (2 * theta + sin(2 * theta) - pi * sin(lat)) /
      (2 + 2 * cos(2 * theta))
    theta <- theta - step
    if (max(abs(step), na.rm = TRUE) < 1e-12) break
  }

  x <- (2 * sqrt(2) / pi) * lon * cos(theta)
  y <- sqrt(2) * sin(theta)

  cbind(x = x, y = y)
}

## Graticule for reference
make_graticule <- function(lons = seq(-150, 150, by = 30),
                           lats = seq(-60, 60, by = 30),
                           n_grid = 400) {
  out <- list()
  idx <- 1

  ## latitude lines
  for (lat0 in lats) {
    lon_seq <- seq(-180, 180, length.out = n_grid)

    ## reverse longitude for astronomical convention
    xy_lat <- mollweide(-lon_seq, rep(lat0, n_grid))

    out[[idx]] <- data.frame(
      x = xy_lat[, "x"],
      y = xy_lat[, "y"],
      group = paste0("lat_", lat0)
    )
    idx <- idx + 1
  }

  ## longitude lines
  for (lon0 in lons) {
    lat_seq <- seq(-89.9, 89.9, length.out = n_grid)

    ## reverse longitude for astronomical convention
    xy_lon <- mollweide(rep(-lon0, n_grid), lat_seq)

    out[[idx]] <- data.frame(
      x = xy_lon[, "x"],
      y = xy_lon[, "y"],
      group = paste0("lon_", lon0)
    )
    idx <- idx + 1
  }

  bind_rows(out)
}
